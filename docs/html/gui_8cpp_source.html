134std::expected<void, Subsystem::Error> Gui::initialize()
135{
136 if (this->is_initialized()) return {};
137
138 IMGUI_CHECKVERSION();
139 ImGui::CreateContext();
140 ImGuiIO &io = ImGui::GetIO();
141
142 // Enable Keyboard and Gamepad Controls
143 io.ConfigFlags |= ImGuiConfigFlags_NavEnableKeyboard;
144 io.ConfigFlags |= ImGuiConfigFlags_NavEnableGamepad;
145 io.ConfigFlags |= ImGuiConfigFlags_DockingEnable;
146
147 // Setup Platform/Renderer backends
148 ImGui_ImplGlfw_InitForOpenGL((GLFWwindow*)Window::get_window(), true);
149 ImGui_ImplOpenGL3_Init();
150 ImGui::SetNextWindowPos(ImVec2(0, 0));
151 setup_style();
152
153 Gui::initialized = true;
154 INFO("{} initialized", Gui::subsystem_name);
155 return {};
156}
157
158std::expected<void, Subsystem::Error> Gui::terminate()
159{
160 if (!this->is_initialized()) return {};
161
162 ImGui_ImplOpenGL3_Shutdown();
163 ImGui_ImplGlfw_Shutdown();
164 ImGui::DestroyContext();
165
166 Gui::initialized = false;
167 INFO("{}: terminated", Gui::subsystem_name);
168 return {};
169}

207void Gui::new_frame(FrameBuffer *fb, std::string name)
208{
209 ImGui_ImplOpenGL3_NewFrame();
210 ImGui_ImplGlfw_NewFrame();
211 ImGui::NewFrame();
212
213 ImGui::DockSpaceOverViewport(0, ImGui::GetMainViewport());
214
215 //
216 // Game window
217 //
218
219 ImGui::SetNextWindowSize(ImVec2(Window::get_width() * 0.8,
220 Window::get_width() * 0.8),
221 ImGuiCond_FirstUseEver);
222 ImGui::Begin(name.c_str());
223
224 float window_width = ImGui::GetContentRegionAvail().x;
225 float window_height = ImGui::GetContentRegionAvail().y;
226
227 fb->rescale(window_width, window_height);
228 Gl::set_viewport(0, 0, window_width, window_height);
229
230 ImGui::Image((void *) (intptr_t) fb->texture_id,
231 ImGui::GetContentRegionAvail(), ImVec2(0, 1), ImVec2(1, 0));
232
233 ImGui::End();
234 return;
235}
