45void Model::init()
46{
47 // Load with assimp
48 Assimp::Importer importer;
49 const aiScene *scene =
50 importer.ReadFile(this->path, aiProcess_Triangulate | aiProcess_FlipUVs);
51
52 if (!scene || scene->mFlags & AI_SCENE_FLAGS_INCOMPLETE || !scene->mRootNode)
53 {
54 ERROR("model::load_model: Could not load model with assimp: {}",
55 importer.GetErrorString());
56 return;
57 }
58 this->directory = this->path.parent_path();
59
60 process_node(scene->mRootNode, scene);
61 return;
62}

73void Model::process_node(aiNode *node, const aiScene *scene)
74{
75 for (unsigned int i = 0; i < node->mNumMeshes; i++)

77 aiMesh *m = scene->mMeshes[node->mMeshes[i]];
78 process_mesh(m, scene);
79 }
80 for (unsigned int i = 0; i < node->mNumChildren; i++)
81 {
82 process_node(node->mChildren[i], scene);
83 }
84 return;

87void Model::process_mesh(aiMesh *m, const aiScene *scene)
88{
89 std::vector<Mesh::Vertex> vertices;
90 std::vector<unsigned int> indices;
91 std::vector<std::shared_ptr<Texture>> textures;
92
93 for (unsigned int i = 0; i < m->mNumVertices; i++)
94 {
95 Mesh::Vertex vertex;
96 glm::vec3 vector;
97 vector.x = m->mVertices[i].x;
98 vector.y = m->mVertices[i].y;
99 vector.z = m->mVertices[i].z;
100
101 vertex.position = vector;
102
103 vector.x = m->mNormals[i].x;
104 vector.y = m->mNormals[i].y;
105 vector.z = m->mNormals[i].z;
106 vertex.normal = vector;
107
108 if (m->mTextureCoords[0])
109 {
110 glm::vec2 vec;
111 vec.x = m->mTextureCoords[0][i].x;
112 vec.y = m->mTextureCoords[0][i].y;
113 vertex.tex_coords = vec;
114 }
115 else
116 vertex.tex_coords = glm::vec2(0.0f, 0.0f);
117
118 vertices.push_back(vertex);
119 }
120
121 for (unsigned int i = 0; i < m->mNumFaces; i++)
122 {
123 aiFace face = m->mFaces[i];
124 for (unsigned int j = 0; j < face.mNumIndices; j++)
125 indices.push_back(face.mIndices[j]);
126 }
127
128 aiMaterial *material = scene->mMaterials[m->mMaterialIndex];
129 std::vector<std::shared_ptr<Texture>> diffuse =
130 load_material_textures(material,
131 aiTextureType_DIFFUSE,
132 Texture::Type::Diffuse);
133 textures.insert(textures.end(),
134 diffuse.begin(),
135 diffuse.end());
136
137 std::vector<std::shared_ptr<Texture>> specular =
138 load_material_textures(material,
139 aiTextureType_SPECULAR,
140 Texture::Type::Specular);
141 textures.insert(textures.end(),
142 specular.begin(),
143 specular.end());
144
145 Mesh mesh = Mesh({vertices, indices, textures, this->wrapping,
146 this->filtering_min, this->filtering_mag, this->has_mipmap,
147 this->mipmap_min, this->mipmap_mag});
148 meshes.push_back(std::move(mesh));
149}

151std::vector<std::shared_ptr<Texture>>
152Model::load_material_textures(aiMaterial *mat,
153 aiTextureType type,
154 Texture::Type type_brenta)
155{
156 std::vector<std::shared_ptr<Texture>> textures;
157 for (unsigned int i = 0; i < mat->GetTextureCount(type); i++)
158 {
159 aiString str;
160 mat->GetTexture(type, i, &str);
161 bool skip = false;
162 std::string path =
163 std::string(this->directory.c_str()) + "/" + std::string(str.C_Str());
164
165 for (unsigned int j = 0; j < textures_loaded.size(); j++)
166 {
167 // Do not load the same texture again
168 if (textures_loaded[j]->path == path)
169 {
170 textures.push_back(textures_loaded[j]);
171 skip = true;
172 break;
173 }
174 }
175 if (!skip)
176 {
177 Texture t = Texture(path, this->flip, type_brenta);
178 std::shared_ptr<Texture> t_ptr = std::make_shared<Texture>(std::move(t));
179 textures_loaded.push_back(t_ptr);
180 textures.push_back(t_ptr);
181 }
182 }
183 return textures;
184}
