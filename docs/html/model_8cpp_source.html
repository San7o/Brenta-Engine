58void Model::load(const Texture::Properties &props)
59{
60 tinyobj::ObjReaderConfig reader_config;
61 std::string path_str = path.string();
62 this->directory = path_str.substr(0, path_str.find_last_of('/'));
63 reader_config.mtl_search_path = this->directory; // Path to material files
64
65 tinyobj::ObjReader reader;
66
67 if (!reader.ParseFromFile(this->path, reader_config))
68 {
69 if (!reader.Error().empty())
70 {
71 std::cerr << "TinyObjReader: " << reader.Error();
72 }
73 exit(1);
74 }
75
76 auto& attrib = reader.GetAttrib();
77 auto& shapes = reader.GetShapes();
78 auto& materials = reader.GetMaterials();
79
80 // Loop over shapes (equivalent to Assimp meshes/nodes)
81 for (size_t s = 0; s < shapes.size(); s++)
82 {
83 process_shape(attrib, shapes[s], materials, props);
84 }
85}

87void Model::process_shape(const tinyobj::attrib_t& attrib,
88 const tinyobj::shape_t& shape,
89 const std::vector<tinyobj::material_t>& materials,
90 const Texture::Properties& props)
91{
92 // Map material_id -> Mesh Data (vertices and indices)
93 std::map<int, std::vector<Mesh::Vertex>> per_mat_vertices;
94 std::map<int, std::vector<unsigned int>> per_mat_indices;
95
96 size_t index_offset = 0;
97
98 for (size_t f = 0; f < shape.mesh.num_face_vertices.size(); f++)
99 {
100 size_t fv = size_t(shape.mesh.num_face_vertices[f]);
101 int mat_id = shape.mesh.material_ids[f]; // Get material for THIS face
102
103 for (size_t v = 0; v < fv; v++)
104 {
105 tinyobj::index_t idx = shape.mesh.indices[index_offset + v];
106 Mesh::Vertex vertex;
107
108 // Positions
109 vertex.position = {
110 attrib.vertices[3 * idx.vertex_index + 0],
111 attrib.vertices[3 * idx.vertex_index + 1],
112 attrib.vertices[3 * idx.vertex_index + 2]
113 };
114
115 // Normals
116 if (idx.normal_index >= 0)
117 {
118 vertex.normal = {
119 attrib.normals[3 * idx.normal_index + 0],
120 attrib.normals[3 * idx.normal_index + 1],
121 attrib.normals[3 * idx.normal_index + 2]
122 };
123 }
124
125 // UVs with Flip
126 if (idx.texcoord_index >= 0)
127 {
128 vertex.tex_coords = {
129 attrib.texcoords[2 * idx.texcoord_index + 0],
130 1.0f - attrib.texcoords[2 * idx.texcoord_index + 1]
131 };
132 }
133
134 per_mat_vertices[mat_id].push_back(vertex);
135 per_mat_indices[mat_id].push_back(per_mat_indices[mat_id].size());
136 }
137 index_offset += fv;
138 }
139
140 // Now create a Mesh for each material group found in this shape
141 for (auto const& [mat_id, verts] : per_mat_vertices)
142 {
143 std::vector<std::shared_ptr<Texture>> textures;
144 if (mat_id >= 0)
145 {
146 textures = load_tiny_material(materials[mat_id], props);
147 }
148
149 // Push the mesh to your model's mesh list
150 meshes.push_back(Mesh({verts, per_mat_indices[mat_id], textures}));
151 }
152}

154std::vector<std::shared_ptr<Texture>>
155Model::load_tiny_material(const tinyobj::material_t& mat,
156 const Texture::Properties &props)
157{
158 std::vector<std::shared_ptr<Texture>> textures;
159
160 auto load_tex = [&](std::string tex_name, Texture::Type type)
161 {
162 if (tex_name.empty()) return;
163
164 std::string full_path = this->directory + "/" + tex_name;
165
166 // Check cache (textures_loaded)
167 for (auto& loaded : textures_loaded)
168 {
169 if (loaded->get_path() == full_path)
170 {
171 textures.push_back(loaded);
172 return;
173 }
174 }
175
176 // Use your Builder pattern
177 auto t =
178 std::make_shared<Texture>(Texture::Builder()
179 .type(type)
180 .target(Texture::Target::Texture2D)
181 .path(full_path)
182 .properties(props)
183 .build());
184
185 textures_loaded.push_back(t);
186 textures.push_back(t);
187 };
188
189 load_tex(mat.diffuse_texname, Texture::Type::Diffuse);
190 load_tex(mat.specular_texname, Texture::Type::Specular);
191
192 return textures;
193}
