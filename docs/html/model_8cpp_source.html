34void model::load_model(std::string path)
35{
36 /* Load with assimp */
37 Assimp::Importer importer;
38 const aiScene *scene =
39 importer.ReadFile(path, aiProcess_Triangulate | aiProcess_FlipUVs);
40
41 if (!scene || scene->mFlags & AI_SCENE_FLAGS_INCOMPLETE || !scene->mRootNode)
42 {
43 ERROR("Could not load model with assimp: {}", importer.GetErrorString());
44 return;
45 }
46 directory = path.substr(0, path.find_last_of('/'));
47
48 process_node(scene->mRootNode, scene);
49}

58 for (unsigned int i = 0; i < node->mNumChildren; i++)

64mesh model::process_mesh(aiMesh *m, const aiScene *scene)
65{
66 std::vector<types::vertex> vertices;
67 std::vector<unsigned int> indices;
68 std::vector<types::texture> textures;
69
70 for (unsigned int i = 0; i < m->mNumVertices; i++)
71 {
72 types::vertex vertex;
73 glm::vec3 vector;
74 vector.x = m->mVertices[i].x;
75 vector.y = m->mVertices[i].y;
76 vector.z = m->mVertices[i].z;
77 vertex.position = vector;
78
79 vector.x = m->mNormals[i].x;
80 vector.y = m->mNormals[i].y;
81 vector.z = m->mNormals[i].z;
82 vertex.normal = vector;
83
84 if (m->mTextureCoords[0])
85 {
86 glm::vec2 vec;
87 vec.x = m->mTextureCoords[0][i].x;
88 vec.y = m->mTextureCoords[0][i].y;
89 vertex.tex_coords = vec;
90 }
91 else
92 vertex.tex_coords = glm::vec2(0.0f, 0.0f);
93
94 vertices.push_back(vertex);
95 }
96
97 for (unsigned int i = 0; i < m->mNumFaces; i++)
98 {
99 aiFace face = m->mFaces[i];
100 for (unsigned int j = 0; j < face.mNumIndices; j++)
101 indices.push_back(face.mIndices[j]);
102 }
103
104 aiMaterial *material = scene->mMaterials[m->mMaterialIndex];
105 std::vector<types::texture> diffuseMaps =
106 load_material_textures(material, aiTextureType_DIFFUSE, "texture_diffuse");
107 textures.insert(textures.end(), diffuseMaps.begin(), diffuseMaps.end());
108 std::vector<types::texture> specularMaps = load_material_textures(
109 material, aiTextureType_SPECULAR, "texture_specular");
110 textures.insert(textures.end(), specularMaps.begin(), specularMaps.end());
111
112 return mesh(vertices, indices, textures, this->wrapping, this->filtering_min,
113 this->filtering_mag, this->has_mipmap, this->mipmap_min,
114 this->mipmap_mag);
115}

117std::vector<types::texture> model::load_material_textures(aiMaterial *mat,
118 aiTextureType type,
119 std::string typeName)
120{
121 std::vector<types::texture> textures;
122 for (unsigned int i = 0; i < mat->GetTextureCount(type); i++)
123 {
124 aiString str;
125 mat->GetTexture(type, i, &str);
126 bool skip = false;
127 for (unsigned int j = 0; j < textures_loaded.size(); j++)
128 {
129 if (std::strcmp(textures_loaded[j].path.c_str(), str.C_Str()) == 0)
130 {
131 textures.push_back(textures_loaded[j]);
132 skip = true;
133 break;
134 }
135 }
136 if (!skip)
137 {
138 types::texture texture;
139 std::string path = directory + "/" + std::string(str.C_Str());
140 texture.id = texture::load_texture(path, this->flip);
141 texture.type = typeName;
142 texture.path = str.C_Str();
143 textures.push_back(texture);
144 textures_loaded.push_back(texture);
145 }
146 }
147 return textures;
148}
