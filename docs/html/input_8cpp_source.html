29void input::remove_keyboard_callback(int key)
30{
31 if (input::keyboard_callbacks.find(key) == input::keyboard_callbacks.end())
32 {
33 ERROR("No callback found for key: {}", key);
34 return;
35 }
36
37 input::keyboard_callbacks.erase(key);

41void input::key_callback([[maybe_unused]] GLFWwindow *window,
42 [[maybe_unused]] int key,
43 [[maybe_unused]] int scancode, int action,
44 [[maybe_unused]] int mods)
45{
