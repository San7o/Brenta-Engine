14std::optional<Shader>
15Shader::create(const Shader::Name &shader_name,
16 const std::vector<Shader::Object> &objects)
17{
18 std::vector<Shader::Id> compiled_shaders = {};
19 if (!compile_shaders(compiled_shaders, objects))
20 {
21 ERROR("shader: error compiling shader {}", shader_name);
22 return {};
23 }
24
25 std::optional<Shader::Id> id =
26 Shader::link_program(compiled_shaders, nullptr, 0);
27 if (!id) return {};
28
29 Shader::clean_compilation(compiled_shaders);
30 Shader::shaders.insert({shader_name, *id});
31 return Shader(*id, shader_name);
32}

55bool Shader::compile_shaders(std::vector<Shader::Id> &compiled,
56 const std::vector<Shader::Object> &objects)
57{
58 for (auto& obj : objects)
59 {
60 GLenum shader_type_gl;
61 switch(obj.type)
62 {
63 case Shader::Type::Fragment: shader_type_gl = GL_FRAGMENT_SHADER; break;
64 case Shader::Type::Vertex: shader_type_gl = GL_VERTEX_SHADER; break;
65 case Shader::Type::Geometry: shader_type_gl = GL_GEOMETRY_SHADER; break;
66 case Shader::Type::Compute: shader_type_gl = GL_COMPUTE_SHADER; break;
67 default: shader_type_gl = 0; break;
68 }
69
70 unsigned int shader = glCreateShader(shader_type_gl);
71 const GLchar *src = obj.src.c_str();
72 glShaderSource(shader, 1, &src, NULL);
73 glCompileShader(shader);
74 if (!Shader::check_compile_errors(shader))
75 return false;
76
77 compiled.push_back(shader);
78 }
79 return true;
80}

87std::optional<Shader::Id>
88Shader::link_program(std::vector<Shader::Id>& compiled_shaders,
89 const GLchar **feedback_varyings, int num_varyings)
90{
91 Shader::Id id = glCreateProgram();
92 std::for_each(compiled_shaders.begin(), compiled_shaders.end(),
93 [&id](auto shader) { glAttachShader(id, shader); });
94
95 if (feedback_varyings != nullptr)
96 glTransformFeedbackVaryings(id, num_varyings, feedback_varyings,
97 GL_INTERLEAVED_ATTRIBS);
98
99 glLinkProgram(id);
100 if (!Shader::check_link_errors(id))
101 return {};
102 return id;
103}

105void Shader::clean_compilation(std::vector<Shader::Id>& compiled_shaders)
106{
107 std::for_each(compiled_shaders.begin(), compiled_shaders.end(),
108 [](auto shader) { glDeleteShader(shader); });
109 return;

145bool Shader::set_bool(const GLchar *unif_name, bool value)
146{
147 GLuint program = this->get_id();
148 GLint location = glGetUniformLocation(program, unif_name);
149
150 if (location == -1)
151 {
152 ERROR("Shader::set_bool: uniform '{}' not found in shader '{}'",
153 unif_name, this->get_name());
154 return false;
155 }
156
157 glUniform1i(location, (int) value);
158
159 GLenum err;
160 if ((err = glGetError()) != GL_NO_ERROR)
161 {
162 ERROR("Shader::set_bool: error setting bool value for shader {} with name {}: {}",
163 this->get_name(), unif_name, err);
164 return false;
165 }
166 return true;

337bool Shader::check_compile_errors(Shader::Id shader)
338{
339 int success;
340 char infoLog[1024];
341 glGetShaderiv(shader, GL_COMPILE_STATUS, &success);
342 if (success) return true;
343
344 std::stringstream out;
345 glGetShaderInfoLog(shader, 1024, NULL, infoLog);
346 out << "shader: compilation error: " << infoLog;
347 ERROR("{}", out.str());
348 return false;
349}
350
351bool Shader::check_link_errors(Shader::Id shader)
352{
353 int success;
354 char infoLog[1024];
355 glGetProgramiv(shader, GL_LINK_STATUS, &success);
356 if (success) return true;
357
358 std::stringstream out;
359 glGetProgramInfoLog(shader, 1024, NULL, infoLog);
360 out << "shader: program linking error: " << infoLog;
361 ERROR("{}", out.str());
362 return false;
363}

384std::optional<std::string>
385Shader::Object::read_file(const std::filesystem::path &path)
386{
387 std::string code;
388 std::ifstream file;
389 file.exceptions(std::ifstream::failbit | std::ifstream::badbit);
390
391 try
392 {
393 file.open(path);
394 if (!file.is_open()) throw "Cannot open file";
395 std::stringstream stream;
396 stream << file.rdbuf();
397 file.close();
398 code = stream.str();
399 }
400 catch (std::ifstream::failure &e)
401 {
402 ERROR("shader: error reading shader file: {}", path.c_str());
403 return {};
404 }
405
406 if (code.empty())
407 {
408 ERROR("shader: file is empty: {}", path.c_str());
409 return {};
410 }
411
412 return code;
413}
