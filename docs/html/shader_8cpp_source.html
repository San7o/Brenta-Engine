12unsigned int shader::get_id(types::shader_name_t shader_name)
13{

21/* Use/activate the shader */
22void shader::use(types::shader_name_t shader_name)
23{
24 glUseProgram(shader::get_id(shader_name));
25 GLenum err;
26 if ((err = glGetError()) != GL_NO_ERROR)
27 {
28 ERROR("Error using shader: {}", shader_name);
29 std::cout << err << std::endl;
30 }
31}

105/* utility function for checking shader
106 * compilation/linking errors. */
107void shader::check_compile_errors(unsigned int shader, std::string type)
108{
109 int success;
110 char infoLog[1024];
111 if (type != "PROGRAM")
112 {
113 glGetShaderiv(shader, GL_COMPILE_STATUS, &success);
114 if (!success)
115 {
116 glGetShaderInfoLog(shader, 1024, NULL, infoLog);
117 std::cout << "ERROR::SHADER_COMPILATION_ERROR of type: " << type << "\n"
118 << infoLog << std::endl;
119 }
120 }
121 else
122 {
123 glGetProgramiv(shader, GL_LINK_STATUS, &success);
124 if (!success)
125 {
126 glGetProgramInfoLog(shader, 1024, NULL, infoLog);
127 std::cout << "ERROR::PROGRAM_LINKING_ERROR of type: " << type << "\n"
128 << infoLog << std::endl;
129 }
130 }
