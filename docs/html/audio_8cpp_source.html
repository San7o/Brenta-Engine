18std::expected<void, std::string> audio::initialize()
19{
20 if (!SDL_Init(SDL_INIT_AUDIO))
21 {
22 auto error = SDL_GetError();
23 ERROR("SDL Audio failed to initialize: {}", error);
24 return std::unexpected(this->subsystem_name);
25 }
26
27 for (auto& f : init_files)
28 {
29 load(f.first, f.second);
30 }
31
32 for (auto& s : init_streams)
33 {
34 create_stream(s.first);
35 set_volume(s.first, s.second);
36 }
37
38 create_stream("default");
39 INFO("audio initialized");
40 return {};
41}

43std::expected<void, std::string> audio::terminate()
44{

54 INFO("audio termianted");
55 return {};

64void audio::load(types::audio_name_t name, std::string path)
65{
66 types::audio_file_t audiofile;
67 audiofile.path = path;
68
69 if (!SDL_LoadWAV(path.c_str(), &audiofile.spec, &audiofile.audio_buf,
70 &audiofile.audio_len))
71 {
72 auto error = SDL_GetError();
73 ERROR("SDL Audio failed to load WAV file: {}", error);
74 return;
75 }
76
77 audio::audio_files.insert({name, audiofile});
78 INFO("Loaded audio at {}", path);
79}

81void audio::play_audio(types::audio_name_t audio_name,
82 types::stream_name_t stream_name)
83{
84 auto stream = audio::get_stream(stream_name);
85 if (stream == nullptr)
86 {
87 ERROR("Could not play audio: Audio stream not found");
88 return;
89 }
90
91 clear_stream(stream_name);
92 auto audiofile = audio::get_audio_file(audio_name);
93 if (!SDL_PutAudioStreamData(stream, audiofile.audio_buf, audiofile.audio_len))
94 check_error_audio();
95}

97void audio::create_stream(types::stream_name_t name)
98{
99 SDL_AudioStream *stream = SDL_OpenAudioDeviceStream(
100 SDL_AUDIO_DEVICE_DEFAULT_PLAYBACK, NULL, NULL, NULL);
101 if (stream == NULL)
102 {
103 const char *error = SDL_GetError();
104 ERROR("SDL Audio failed to create stream: {}", error);
105 return;
106 }
107
108 audio::streams.insert({name, stream});
109 resume_stream(name);
110 INFO("SDL Audio stream created");
111}
