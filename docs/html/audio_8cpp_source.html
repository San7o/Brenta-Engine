29void audio::destroy()
30{
31 for (auto &stream : audio::streams)
32 SDL_DestroyAudioStream(stream.second);
33
34 for (auto &audiofile : audio::audio_files)

36 SDL_free(audiofile.second.audio_buf);
37 }
38
39 SDL_Quit();
40 INFO("SDL Audio destroyed");
41}

43void audio::load_audio(types::audio_name_t name, std::string path)
44{
45 types::audio_file_t audiofile;
46 audiofile.path = path;
47
48 if (!SDL_LoadWAV(path.c_str(), &audiofile.spec, &audiofile.audio_buf,
49 &audiofile.audio_len))
50 {
51 auto error = SDL_GetError();
52 ERROR("SDL Audio failed to load WAV file: {}", error);
53 return;
54 }
55
56 audio::audio_files.insert({name, audiofile});
57 INFO("Loaded audio at {}", path);
58}

60void audio::play_audio(types::audio_name_t audio_name,
61 types::stream_name_t stream_name)
62{
63 auto stream = audio::get_stream(stream_name);
64 if (stream == nullptr)
65 {
66 ERROR("Could not play audio: Audio stream not found");
67 return;
68 }
69
70 clear_stream(stream_name);
71 auto audiofile = audio::get_audio_file(audio_name);
72 if (!SDL_PutAudioStreamData(stream, audiofile.audio_buf, audiofile.audio_len))
73 check_error_audio();
74}

76void audio::create_stream(types::stream_name_t name)
77{
78 SDL_AudioStream *stream = SDL_OpenAudioDeviceStream(
79 SDL_AUDIO_DEVICE_DEFAULT_PLAYBACK, NULL, NULL, NULL);
80 if (stream == NULL)
81 {
82 const char *error = SDL_GetError();
83 ERROR("SDL Audio failed to create stream: {}", error);
84 return;
85 }
86
87 audio::streams.insert({name, stream});
88 resume_stream(name);
89 INFO("SDL Audio stream created");
90}
