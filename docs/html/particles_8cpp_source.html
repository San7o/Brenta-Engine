35particle_emitter::particle_emitter(
36 glm::vec3 starting_position, glm::vec3 starting_velocity,
37 glm::vec3 starting_spread, float starting_time_to_live, int num_particles,
38 float spawn_rate, float scale, std::string atlas_path, int atlas_width,
39 int atlas_height, int atlas_index, camera *cam)
40{
41 this->starting_position = starting_position;
42 this->starting_velocity = starting_velocity;
43 this->starting_spread = starting_spread;
44 this->starting_time_to_live = starting_time_to_live;
45 this->num_particles = num_particles;
46 this->spawn_rate = spawn_rate;
47 this->scale = scale;
48 this->atlas = atlas;
49 this->atlas_width = atlas_width;
50 this->atlas_height = atlas_height;
51 this->atlas_index = atlas_index;
52 this->current = 0;
53 this->cam = cam;
54
55 // Load Texture Atlas
56 this->atlas = texture::load_texture(atlas_path, false);
57
58 // Create shaders
59 const GLchar *varyings[] = {"outPosition", "outVelocity", "outTTL"};
60 shader::create(varyings, 3, "particle_update",
61 GL_VERTEX_SHADER, "src/shaders/particle_update.vs");
62 shader::create("particle_render",
63 GL_VERTEX_SHADER, "src/shaders/particle_render.vs",
64 GL_GEOMETRY_SHADER, "src/shaders/particle_render.gs",
65 GL_FRAGMENT_SHADER, "src/shaders/particle_render.fs");
66
67 // This is needed to render points
68 glEnable(GL_PROGRAM_POINT_SIZE);
69
70 this->vao.init();
71 this->vao.bind();
72 check_opengl_error("vao bind");
73
74 // Create fbos
75 this->fbo[0] = types::buffer(GL_TRANSFORM_FEEDBACK_BUFFER);
76 this->fbo[1] = types::buffer(GL_TRANSFORM_FEEDBACK_BUFFER);
77
78 glBindBuffer(GL_TRANSFORM_FEEDBACK_BUFFER, this->fbo[0].id);
79 glBufferData(GL_TRANSFORM_FEEDBACK_BUFFER,
80 this->num_particles * 2 * sizeof(glm::vec3)
81 + this->num_particles * sizeof(float),
82 NULL, GL_DYNAMIC_COPY);
83 glBindBuffer(GL_TRANSFORM_FEEDBACK_BUFFER, this->fbo[1].id);
84 glBufferData(GL_TRANSFORM_FEEDBACK_BUFFER,
85 this->num_particles * 2 * sizeof(glm::vec3)
86 + this->num_particles * sizeof(float),
87 NULL, GL_DYNAMIC_COPY);
88 check_opengl_error("glBindBufferBase A");
89
90 // Unbind buffers
91 glBindBuffer(GL_ARRAY_BUFFER, 0);
92 glBindVertexArray(0);
93 this->vao.unbind();
94}

103// Update particles using Transform Feedback
104// directly from the wiki
105void particle_emitter::update_particles(float delta_time)
106{
107 shader::use("particle_update");
108 shader::set_float("particle_update", "deltaTime", delta_time);
109 shader::set_vec3("particle_update", "emitterPos", this->starting_position);
110 shader::set_vec3("particle_update", "emitterSpread", this->starting_spread);
111 shader::set_float("particle_update", "spawnProbability", this->spawn_rate);
112 shader::set_vec3("particle_update", "emitterVel", this->starting_velocity);
113 shader::set_float("particle_update", "emitterTTL",
114 this->starting_time_to_live);
115 check_opengl_error("settin update shader");
116
117 this->vao.bind();
118
119 glBindBufferBase(GL_TRANSFORM_FEEDBACK_BUFFER, 0, fbo[current].id);
120 check_opengl_error("glBindBufferBase B");
121
122 glBindBuffer(GL_ARRAY_BUFFER, fbo[!current].id);
123 glVertexAttribPointer(0, 3, GL_FLOAT, GL_FALSE,
124 2 * sizeof(glm::vec3) + sizeof(float), (void *) 0);
125 glEnableVertexAttribArray(0);
126 glVertexAttribPointer(1, 3, GL_FLOAT, GL_FALSE,
127 2 * sizeof(glm::vec3) + sizeof(float),
128 (void *) sizeof(glm::vec3));
129 glEnableVertexAttribArray(1);
130 glVertexAttribPointer(2, 1, GL_FLOAT, GL_FALSE,
131 2 * sizeof(glm::vec3) + sizeof(float),
132 (void *) (2 * sizeof(glm::vec3)));
133 glEnableVertexAttribArray(2);
134
135 // Start transform feedback
136 glEnable(GL_RASTERIZER_DISCARD); // Disable rasterization
137 glBeginTransformFeedback(GL_POINTS); // Enter transform feedback mode
138 check_opengl_error("glBeginTransformFeedback");
139
140 glDrawArrays(GL_POINTS, 0, num_particles);
141 check_opengl_error("glDrawTransformFeedback");
142
143 glEndTransformFeedback(); // Exit transform feedback mode
144 glDisable(GL_RASTERIZER_DISCARD); // Enable rasterization
145 // Unbind buffers
146 glBindVertexArray(0);
147 glBindBuffer(GL_ARRAY_BUFFER, 0);
148 glBindBufferBase(GL_TRANSFORM_FEEDBACK_BUFFER, 0, 0);
149 this->vao.unbind();
150 current = !current; // Swap buffers
151}

153// Render particles
154void particle_emitter::render_particles()
155{
156 if (this->cam == nullptr)
157 {
158 ERROR("Camera not set or null for ParticleEmitter");
159 return;
160 }
161
162 shader::use("particle_render");
163
164 this->vao.bind();
165
166 glBindBuffer(GL_ARRAY_BUFFER, fbo[current].id);
167 glVertexAttribPointer(0, 3, GL_FLOAT, GL_FALSE,
168 2 * sizeof(glm::vec3) + sizeof(float), (void *) 0);
169 glEnableVertexAttribArray(0);
170 glVertexAttribPointer(1, 1, GL_FLOAT, GL_FALSE,
171 2 * sizeof(glm::vec3) + sizeof(float),
172 (void *) (2 * sizeof(glm::vec3)));
173 check_opengl_error("glVertexAttribPointer");
174 glEnableVertexAttribArray(1);
175
176 // Set uniforms
177 int window_width = window::get_width();
178 int window_height = window::get_height();
179 types::translation t = types::translation();
180 t.set_view(this->cam->get_view_matrix());
181 t.set_projection(this->cam->get_projection_matrix(window_width, window_height));
182 t.set_model(glm::mat4(1.0f));
183 t.set_shader("particle_render");
184 shader::set_int("particle_render", "atlas_width", this->atlas_width);
185 shader::set_int("particle_render", "atlas_height", this->atlas_height);
186 shader::set_int("particle_render", "atlas_index", this->atlas_index);
187 shader::set_float("particle_render", "scale", this->scale);
188 shader::set_float("particle_render", "aspect_ratio",
189 (float) window_width
190 / (float) window_height);
191
192 // Set Textures
193 texture::active_texture(GL_TEXTURE0);
194 texture::bind_texture(GL_TEXTURE_2D, this->atlas, GL_REPEAT, GL_NEAREST,
195 GL_NEAREST, GL_TRUE, GL_NEAREST_MIPMAP_NEAREST,
196 GL_NEAREST);
197
198 glDrawArrays(GL_POINTS, 0, num_particles);
199 check_opengl_error("glDrawArrays");
200
201 glBindBuffer(GL_ARRAY_BUFFER, 0);
202 glBindVertexArray(0);
203 vao.unbind();
204}
