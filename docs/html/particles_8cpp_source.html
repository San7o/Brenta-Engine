34ParticleEmitter::ParticleEmitter(Config conf)
35{
36 this->starting_position = conf.starting_position;
37 this->starting_velocity = conf.starting_velocity;
38 this->starting_spread = conf.starting_spread;
39 this->starting_time_to_live = conf.starting_time_to_live;
40 this->num_particles = conf.num_particles;
41 this->spawn_rate = conf.spawn_rate;
42 this->scale = conf.scale;
43 this->atlas_width = conf.atlas_width;
44 this->atlas_height = conf.atlas_height;
45 this->atlas_index = conf.atlas_index;
46 this->current = 0;
47 this->cam = conf.cam;
48
49 // Load Texture Atlas
50 this->atlas = std::move(Texture(conf.atlas_path, false));
51
52 // Create shaders
53 const GLchar *varyings[] = {"outPosition", "outVelocity", "outTTL"};
54 Shader::create(varyings, 3, "particle_update",
55 Shader::Type::Vertex, "src/renderer/shaders/particle_update.vs");
56 Shader::create("particle_render",
57 Shader::Type::Vertex, "src/renderer/shaders/particle_render.vs",
58 Shader::Type::Geometry, "src/renderer/shaders/particle_render.gs",
59 Shader::Type::Fragment, "src/renderer/shaders/particle_render.fs");
60
61 // This is needed to render points
62 glEnable(GL_PROGRAM_POINT_SIZE);
63
64 this->vao.init();
65 this->vao.bind();
66 Gl::check_error();
67
68 // Create fbos
69 this->fbo[0].init(GL_TRANSFORM_FEEDBACK_BUFFER);
70 this->fbo[1].init(GL_TRANSFORM_FEEDBACK_BUFFER);
71
72 glBindBuffer(GL_TRANSFORM_FEEDBACK_BUFFER, this->fbo[0].id);
73 glBufferData(GL_TRANSFORM_FEEDBACK_BUFFER,
74 this->num_particles * 2 * sizeof(glm::vec3)
75 + this->num_particles * sizeof(float),
76 NULL, GL_DYNAMIC_COPY);
77 glBindBuffer(GL_TRANSFORM_FEEDBACK_BUFFER, this->fbo[1].id);
78 glBufferData(GL_TRANSFORM_FEEDBACK_BUFFER,
79 this->num_particles * 2 * sizeof(glm::vec3)
80 + this->num_particles * sizeof(float),
81 NULL, GL_DYNAMIC_COPY);
82 Gl::check_error();
83
84 // Unbind buffers
85 glBindBuffer(GL_ARRAY_BUFFER, 0);
86 glBindVertexArray(0);
87 this->vao.unbind();
88}

90// Update particles using Transform Feedback
91void ParticleEmitter::update(float delta_time)
92{
93 Shader::use("particle_update");
94 Shader::set_float("particle_update", "deltaTime", delta_time);
95 Shader::set_vec3("particle_update", "emitterPos", this->starting_position);
96 Shader::set_vec3("particle_update", "emitterSpread", this->starting_spread);
97 Shader::set_float("particle_update", "spawnProbability", this->spawn_rate);
98 Shader::set_vec3("particle_update", "emitterVel", this->starting_velocity);
99 Shader::set_float("particle_update", "emitterTTL",
100 this->starting_time_to_live);
101 Gl::check_error();
102
103 this->vao.bind();
104
105 glBindBufferBase(GL_TRANSFORM_FEEDBACK_BUFFER, 0, fbo[current].id);
106 Gl::check_error();
107
108 glBindBuffer(GL_ARRAY_BUFFER, fbo[!current].id);
109 glVertexAttribPointer(0, 3, GL_FLOAT, GL_FALSE,
110 2 * sizeof(glm::vec3) + sizeof(float), (void *) 0);
111 glEnableVertexAttribArray(0);
112 glVertexAttribPointer(1, 3, GL_FLOAT, GL_FALSE,
113 2 * sizeof(glm::vec3) + sizeof(float),
114 (void *) sizeof(glm::vec3));
115 glEnableVertexAttribArray(1);
116 glVertexAttribPointer(2, 1, GL_FLOAT, GL_FALSE,
117 2 * sizeof(glm::vec3) + sizeof(float),
118 (void *) (2 * sizeof(glm::vec3)));
119 glEnableVertexAttribArray(2);
120
121 // Start transform feedback
122 glEnable(GL_RASTERIZER_DISCARD); // Disable rasterization
123 glBeginTransformFeedback(GL_POINTS); // Enter transform feedback mode
124 Gl::check_error();
125
126 glDrawArrays(GL_POINTS, 0, num_particles);
127 Gl::check_error();
128
129 glEndTransformFeedback(); // Exit transform feedback mode
130 glDisable(GL_RASTERIZER_DISCARD); // Enable rasterization
131 // Unbind buffers
132 glBindVertexArray(0);
133 glBindBuffer(GL_ARRAY_BUFFER, 0);
134 glBindBufferBase(GL_TRANSFORM_FEEDBACK_BUFFER, 0, 0);
135 this->vao.unbind();
136 current = !current; // Swap buffers
137}

139// Render particles
140void ParticleEmitter::render()
141{
142 if (!this->cam)
143 {
144 ERROR("ParticleEmitter::render_particles: Camera not set or null for emitter");
145 return;
146 }
147
148 Shader::use("particle_render");
149
150 this->vao.bind();
151
152 glBindBuffer(GL_ARRAY_BUFFER, fbo[current].id);
153 glVertexAttribPointer(0, 3, GL_FLOAT, GL_FALSE,
154 2 * sizeof(glm::vec3) + sizeof(float), (void *) 0);
155 glEnableVertexAttribArray(0);
156 glVertexAttribPointer(1, 1, GL_FLOAT, GL_FALSE,
157 2 * sizeof(glm::vec3) + sizeof(float),
158 (void *) (2 * sizeof(glm::vec3)));
159 Gl::check_error();
160 glEnableVertexAttribArray(1);
161
162 // Set uniforms
163 int window_width = Window::get_width();
164 int window_height = Window::get_height();
165 Translation t = Translation();
166 t.set_view(this->cam->get_view_matrix());
167 t.set_projection(this->cam->get_projection_matrix(window_width, window_height));
168 t.set_model(glm::mat4(1.0f));
169 t.set_shader("particle_render");
170 Shader::set_int("particle_render", "atlas_width", this->atlas_width);
171 Shader::set_int("particle_render", "atlas_height", this->atlas_height);
172 Shader::set_int("particle_render", "atlas_index", this->atlas_index);
173 Shader::set_float("particle_render", "scale", this->scale);
174 Shader::set_float("particle_render", "aspect_ratio",
175 (float) window_width
176 / (float) window_height);
177
178 // Set Textures
179 Texture::active_texture(GL_TEXTURE0);
180 this->atlas.bind(GL_TEXTURE_2D, GL_REPEAT, GL_NEAREST,
181 GL_NEAREST, GL_TRUE, GL_NEAREST_MIPMAP_NEAREST,
182 GL_NEAREST);
183
184 glDrawArrays(GL_POINTS, 0, num_particles);
185 Gl::check_error();
186
187 glBindBuffer(GL_ARRAY_BUFFER, 0);
188 glBindVertexArray(0);
189 vao.unbind();
190 return;
191}
