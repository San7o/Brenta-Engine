36ParticleEmitter::ParticleEmitter(Config conf)
37{
38 this->starting_position = conf.starting_position;
39 this->starting_velocity = conf.starting_velocity;
40 this->starting_spread = conf.starting_spread;
41 this->starting_time_to_live = conf.starting_time_to_live;
42 this->num_particles = conf.num_particles;
43 this->spawn_rate = conf.spawn_rate;
44 this->scale = conf.scale;
45 this->atlas_width = conf.atlas_width;
46 this->atlas_height = conf.atlas_height;
47 this->atlas_index = conf.atlas_index;
48 this->current_fbo_index = 0;
49 this->cam = conf.cam;
50
51 // Load Texture Atlas
52 this->atlas =
53 Texture::Builder()
54 .target(Texture::Target::Texture2D)
55 .path(conf.atlas_path)
56 .properties(Texture::Properties()
57 .wrapping(Texture::Wrapping::Repeat)
58 .filtering_min(Texture::Filtering::Nearest)
59 .filtering_mag(Texture::Filtering::Nearest)
60 .has_mipmap(Gl::True)
61 .mipmap_min(Texture::Filtering::NearestMipmapNearest)
62 .mipmap_mag(Texture::Filtering::Nearest))
63 .build();
64
65 // Create shaders
66 const GLchar *varyings[] = {"outPosition", "outVelocity", "outTTL"};
67 this->shader_update =
68 AssetManager::new_asset<Shader>("particle_update_shader",
69 Shader::Builder()
70 .feedback(varyings,
71 sizeof(varyings) / sizeof(varyings[0]))
72 .object({
73 Shader::Type::Vertex,
74 particle_update_vs
75 }));
76 if (!this->shader_update) return;
77
78 this->shader_render = AssetManager::new_asset<Shader>("particole_render_shader",
79 Shader::Builder()
80 .objects({
81 { Shader::Type::Vertex, particle_render_vs },
82 { Shader::Type::Geometry, particle_render_gs },
83 { Shader::Type::Fragment, particle_render_fs }}));
84 if (!this->shader_render) return;
85
86 // This is needed to render points
87 glEnable(GL_PROGRAM_POINT_SIZE);
88
89 this->vao.init();
90 this->vao.bind();
91 Gl::check_error();
92
93 // Create fbos
94 this->fbo[0].init(Buffer::Target::TransformFeedback);
95 this->fbo[1].init(Buffer::Target::TransformFeedback);
96
97 this->fbo[0].bind();
98 this->fbo[0].copy_data(NULL, this->num_particles * 2 * sizeof(glm::vec3)
99 + this->num_particles * sizeof(float),
100 Buffer::DataUsage::DynamicCopy);
101 this->fbo[1].bind();
102 this->fbo[1].copy_data(NULL,
103 this->num_particles * 2 * sizeof(glm::vec3)
104 + this->num_particles * sizeof(float),
105 Buffer::DataUsage::DynamicCopy);
106 Gl::check_error();
107
108 // Unbind buffers
109 this->fbo[0].unbind();
110 this->vao.unbind();
111
112 // Setup UBO
113 this->ubo.init(*this->shader_update, "settings", 3, sizeof(ParticleSettings));
114 this->ubo.bind();
115 this->ubo.copy_data(NULL,
116 sizeof(ParticleSettings),
117 Buffer::DataUsage::DynamicDraw);
118 this->ubo.unbind();
119
120 return;
121}

123// Update particles using Transform Feedback
124void ParticleEmitter::update(float delta_time)
125{
126 if (!this->shader_update) return;
127 this->shader_update->use();
128
129 ParticleSettings settings = {
130 .gravity = glm::vec3(0.0f, -9.81f, 0.0f),
131 .deltaTime = delta_time,
132 .emitterVel = this->starting_velocity,
133 .emitterTTL = this->starting_time_to_live,
134 .emitterPos = this->starting_position,
135 .spawnProbability = this->spawn_rate,
136 .emitterSpread = this->starting_spread,
137 .__padding = 0.0f,
138 };
139
140 // ubo
141
142 this->ubo.bind();
143 this->ubo.copy_data(&settings, sizeof(ParticleSettings),
144 Buffer::DataUsage::DynamicDraw);
145 this->ubo.unbind();
146 Gl::check_error();
147
148 // Vao
149
150 this->vao.bind();
151 glBindBufferBase(GL_TRANSFORM_FEEDBACK_BUFFER, 0, fbo[current_fbo_index].get_id());
152 Gl::check_error();
153 glBindBuffer(GL_ARRAY_BUFFER, fbo[!current_fbo_index].get_id());
154 this->vao.link_buffer(this->fbo[!current_fbo_index],
155 0, 3, Gl::Float, Gl::False,
156 2 * sizeof(glm::vec3) + sizeof(float), (void *) 0);
157 this->vao.link_buffer(this->fbo[!current_fbo_index],
158 1, 3, Gl::Float, Gl::False,
159 2 * sizeof(glm::vec3) + sizeof(float),
160 (void *) sizeof(glm::vec3));
161 this->vao.link_buffer(this->fbo[!current_fbo_index],
162 2, 1, Gl::Float, Gl::False,
163 2 * sizeof(glm::vec3) + sizeof(float),
164 (void *) (2 * sizeof(glm::vec3)));
165
166 // Start transform feedback
167 glEnable(GL_RASTERIZER_DISCARD); // Disable rasterization
168 glBeginTransformFeedback(GL_POINTS); // Enter transform feedback mode
169 Gl::check_error();
170
171 glDrawArrays(GL_POINTS, 0, num_particles);
172 Gl::check_error();
173
174 glEndTransformFeedback(); // Exit transform feedback mode
175 glDisable(GL_RASTERIZER_DISCARD); // Enable rasterization
176
177 // Unbind buffers
178 glBindBuffer(GL_ARRAY_BUFFER, 0);
179 this->vao.unbind();
180 glBindBufferBase(GL_TRANSFORM_FEEDBACK_BUFFER, 0, 0);
181 return;
182}

184// Render particles
185void ParticleEmitter::render()
186{
187 tenno::shared_ptr<Camera> camera = this->cam.lock();
188 if (!camera)
189 {
190 ERROR("ParticleEmitter::render_particles: Camera not set or null for emitter");
191 return;
192 }
193
194 if (!this->shader_render) return;
195
196 this->vao.bind();
197
198 glBindBuffer(GL_ARRAY_BUFFER, fbo[current_fbo_index].get_id());
199 this->vao.link_buffer(this->fbo[current_fbo_index],
200 0, 3, Gl::Float, Gl::False,
201 2 * sizeof(glm::vec3) + sizeof(float), (void *) 0);
202 this->vao.link_buffer(this->fbo[current_fbo_index],
203 1, 1, Gl::Float, Gl::False,
204 2 * sizeof(glm::vec3) + sizeof(float),
205 (void *) (2 * sizeof(glm::vec3)));
206 Gl::check_error();
207
208 // Set uniforms
209 int window_width = Window::get_width();
210 int window_height = Window::get_height();
211 auto projection =
212 camera->get_projection_matrix(window_width, window_height);
213
214 shader_render->use();
215 shader_render->set_mat4("view", camera->get_view_matrix());
216 shader_render->set_mat4("projection", projection);
217 shader_render->set_mat4("model", glm::mat4(1.0f));
218 shader_render->set_int("atlas_width", this->atlas_width);
219 shader_render->set_int("atlas_height", this->atlas_height);
220 shader_render->set_int("atlas_index", this->atlas_index);
221 shader_render->set_float("scale", this->scale);
222 shader_render->set_float("aspect_ratio",
223 (float) window_width / (float) window_height);
224
225 // Set Textures
226 Texture::active_texture(0);
227 this->atlas.bind();
228
229 glDrawArrays(GL_POINTS, 0, num_particles);
230 Gl::check_error();
231
232 glBindBuffer(GL_ARRAY_BUFFER, 0);
233 this->vao.unbind();
234
235 // Swap buffers
236 current_fbo_index = !current_fbo_index; // Swap buffers
237 return;
238}
