14Font::Font(const std::filesystem::path &path, int size)
15{
16 this->vao.init();
17 this->vao.bind();
18 this->vbo.init(Buffer::Target::Array);
19
20 FT_Library ft;
21 if (FT_Init_FreeType(&ft))
22 {
23 ERROR("Font: could not init FreeType library");
24 return;
25 }
26
27 auto text_shader = AssetManager::get<Shader>("TextShader");
28 if (!text_shader)
29 {
30 text_shader =
31 AssetManager::new_asset<Shader>("TextShader",
32 Shader::Builder()
33 .object({ Shader::Type::Vertex, text_vs })
34 .object({ Shader::Type::Fragment, text_fs }));
35 }
36 if (!text_shader) return;
37
38 this->shader = text_shader;
39 this->shader->use();
40
41 // find path to font
42 if (path.empty())
43 {
44 ERROR("Font: could not find font at path {}", path.string());
45 return;
46 }
47
48 FT_Face face;
49 if (FT_New_Face(ft, path.string().c_str(), 0, &face))
50 {
51 ERROR("Font: could not load font at path {}", path.string());
52 return;
53 }
54 else
55 {
56 // set size to load glyphs as
57 FT_Set_Pixel_Sizes(face, 0, size);
58
59 // disable byte-alignment restriction
60 glPixelStorei(GL_UNPACK_ALIGNMENT, 1);
61
62 // load first 128 characters of ASCII set
63 for (unsigned char c = 0; c < 128; c++)
64 {
65 // Load character glyph
66 if (FT_Load_Char(face, c, FT_LOAD_RENDER))
67 {
68 ERROR("Font: could not load glyph for font at path {}",
69 path.string());
70 continue;
71 }
72 // generate texture
73 unsigned int texture;
74 glGenTextures(1, &texture);
75 Texture::bind_id(Texture::Target::Texture2D, texture);
76 glTexImage2D(GL_TEXTURE_2D, 0, GL_RED, face->glyph->bitmap.width,
77 face->glyph->bitmap.rows, 0, GL_RED, GL_UNSIGNED_BYTE,
78 face->glyph->bitmap.buffer);
79 // set texture options
80 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_CLAMP_TO_EDGE);
81 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_CLAMP_TO_EDGE);
82 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_LINEAR);
83 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_LINEAR);
84 // now store character for later use
85 Font::Character character = {
86 texture,
87 glm::ivec2(face->glyph->bitmap.width, face->glyph->bitmap.rows),
88 glm::ivec2(face->glyph->bitmap_left, face->glyph->bitmap_top),
89 static_cast<unsigned int>(face->glyph->advance.x)};
90 characters.insert(std::pair<char, Font::Character>(c, character));
91 }
92 Texture::bind_id(Texture::Target::Texture2D, 0);
93 }
94 // destroy FreeType once we're finished
95 FT_Done_Face(face);
96 FT_Done_FreeType(ft);
97
98 // configure VAO/VBO for texture quads
99 this->vao.bind();
100 this->vbo.bind();
101 glBufferData(GL_ARRAY_BUFFER, sizeof(float) * 6 * 4, NULL, GL_DYNAMIC_DRAW);
102 glEnableVertexAttribArray(0);
103 glVertexAttribPointer(0, 4, GL_FLOAT, GL_FALSE, 4 * sizeof(float), 0);
104 this->vbo.unbind();
105 this->vao.unbind();
106
107 EVENT(Logger::Event::Lifetime, "Font: initialized");
108}
