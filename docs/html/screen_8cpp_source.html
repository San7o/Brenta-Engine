19void screen::init(int SCR_WIDTH, int SCR_HEIGHT, bool is_mouse_captured,
20 const char *title, bool msaa, bool vsync)
21{
22 screen::WIDTH = SCR_WIDTH;
23 screen::HEIGHT = SCR_HEIGHT;
24
25 if (glfwInit() == GLFW_FALSE)
26 {
27 ERROR("Failed to initialize GLFW on init");
28 return;
29 }
30
31 set_context_version(3, 3);
32 use_core_profile();
33
34 if (msaa)
35 {
36 glfwWindowHint(GLFW_SAMPLES, 4);
37 INFO("Enabled MSAA");
38 }
39
40 if (!vsync)
41 {
42 glfwSwapInterval(0);
43 INFO("Disabled VSync");
44 }
45
46#ifdef __APPLE__
47 set_hints_apple();
48#endif
49
50 create_window(SCR_WIDTH, SCR_HEIGHT, title);
51 make_context_current();
52 set_mouse_capture(is_mouse_captured);
53
54 // Set the callback for resizing the window
55 screen::set_size_callback(framebuffer_size_callback);
56}
