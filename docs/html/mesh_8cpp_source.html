34void Mesh::init()
35{
36 if (this->vertices.size() == 0 || this->indices.size() == 0)
37 return;
38
39 this->vao.init();
40 this->vao.bind();
41
42 this->vbo.init(Buffer::Target::Array);
43 this->vbo.bind();
44
45 this->vbo.copy_data(&this->vertices[0],
46 this->vertices.size() * sizeof(Vertex),
47 Buffer::DataUsage::StaticDraw);
48
49 this->vao.link_buffer(this->vbo, 0, 3, Gl::Type::Float, Gl::False,
50 sizeof(Vertex), (void *) 0);
51 this->vao.link_buffer(this->vbo, 1, 3, Gl::Type::Float, Gl::False,
52 sizeof(Vertex),
53 (void *) offsetof(Vertex, normal));
54 this->vao.link_buffer(this->vbo, 2, 2, Gl::Type::Float, Gl::False,
55 sizeof(Vertex),
56 (void *) offsetof(Vertex, tex_coords));
57
58 this->ebo.init(Buffer::Target::ElementArray);
59 this->ebo.bind();
60 this->ebo.copy_data(&this->indices[0],
61 this->indices.size() * sizeof(unsigned int),
62 Buffer::DataUsage::StaticDraw);
63 this->vao.unbind();
64 this->vbo.unbind();
65 this->ebo.unbind();
66
67 EVENT(Logger::Event::Lifetime, "mesh: created {}", this->id);
68 return;
69}

71void Mesh::draw() const
72{
73 if (this->vao.get_id() == 0)
74 {
75 ERROR("Mesh::draw: not initialized");
76 return;
77 }
78
79 unsigned int diffuseNr = 1;
80 unsigned int specularNr = 1;
81 for (unsigned int i = 0; i < this->textures.size(); i++)
82 {
83 Texture::active_texture(i);
84
85 std::string number;
86 std::string name;
87 auto type = textures[i]->get_type();
88 switch (type)
89 {
90 case Texture::Type::Diffuse:
91 name = "texture_diffuse";
92 number = std::to_string(diffuseNr++);
93 break;
94 case Texture::Type::Specular:
95 name = "texture_specular";
96 number = std::to_string(specularNr++);
97 break;
98 default:
99 number = "0";
100 break;
101 }
102
103 // Get the current shader
104 GLint prog = 0;
105 glGetIntegerv(GL_CURRENT_PROGRAM, &prog);
106
107 // Set uniforms
108 Shader::set_int(prog, ("material." + name + number).c_str(), i);
109 textures[i]->bind();
110 }
111
112 Texture::active_texture(0);
113
114 // draw mesh
115 this->vao.bind();
116 Gl::draw_elements(GL_TRIANGLES, this->indices.size(), GL_UNSIGNED_INT, 0);
117 this->vao.unbind();
118
119 Texture::active_texture(0);
120 return;
121}
