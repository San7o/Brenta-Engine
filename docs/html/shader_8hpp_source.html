138 template <typename... Args>
139 static void compile_shaders(std::vector<unsigned int> &compiled, GLenum type,
140 std::string path, Args... args)
141 {
142 std::string code;
143 std::ifstream file;
144 file.exceptions(std::ifstream::failbit | std::ifstream::badbit);
145
146 try
147 {
148 file.open(path);
149 std::stringstream stream;
150 stream << file.rdbuf();
151 file.close();
152 code = stream.str();
153 }
154 catch (std::ifstream::failure &e)
155 {
156 ERROR("Error reading shader file: {}", path);
157 return;
158 }
159
160 const char *shader_code = code.c_str();
161 unsigned int shader = glCreateShader(type);
162 glShaderSource(shader, 1, &shader_code, NULL);
163 glCompileShader(shader);
164 shader::check_compile_errors(shader, "SHADER");
165
166 compiled.push_back(shader);
167 compile_shaders(compiled, args...);
168 }
