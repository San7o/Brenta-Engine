38std::expected<void, Subsystem::Error> Gl::initialize()
39{
40 if (this->is_initialized()) return {};
41
42 GLADloadproc loadproc = (GLADloadproc) Window::get_proc_address();
43 if (!gladLoadGLLoader(loadproc))
44 {
45 ERROR("{}: failed to initialize GLAD", Gl::subsystem_name);
46 return std::unexpected("Failed to initialize GLAD");
47 }
48
49 int width = Window::get_width();
50 int height = Window::get_height();
51
52 glViewport(0, 0, width, height);
53
54 if (Gl::init_config.enable_depth_test)
55 {
56 glEnable(GL_DEPTH_TEST);
57 INFO("{}: enabled GL_DEPTH_TEST", Gl::subsystem_name);
58 }
59
60 if (Gl::init_config.enable_blending)
61 {
62 glEnable(GL_BLEND);
63 glBlendFunc(GL_SRC_ALPHA, GL_ONE_MINUS_SRC_ALPHA);
64 INFO("{}: enabled GL_BLEND (transparency)", Gl::subsystem_name);
65 }
66
67 if (Gl::init_config.enable_backface_culling)
68 {
69 glEnable(GL_CULL_FACE);
70 INFO("{}: enabled GL_CULL_FACE", Gl::subsystem_name);
71 }
72
73 if (Gl::init_config.enable_multisample)
74 {
75 glEnable(GL_MULTISAMPLE);
76 INFO("{}: enabled GL_MULTISAMPLE", Gl::subsystem_name);
77 }
78
79 int flags;
80 glGetIntegerv(GL_CONTEXT_FLAGS, &flags);
81 if (flags & GL_CONTEXT_FLAG_DEBUG_BIT)
82 {
83 // Enable all messages from all sources, of all types, of all severities.
84 glEnable(GL_DEBUG_OUTPUT);
85 glEnable(GL_DEBUG_OUTPUT_SYNCHRONOUS);
86 glDebugMessageCallback(glDebugOutput, nullptr);
87 glDebugMessageControl(GL_DONT_CARE, GL_DONT_CARE, GL_DONT_CARE,
88 0, nullptr, GL_TRUE);
89 INFO("{}: configured GL_DEBUG_OUTPUT", Gl::subsystem_name);
90 }
91
92 GLenum errcode = Gl::check_error();
93 if (errcode != GL_NO_ERROR)
94 {
95 return std::unexpected("GL error");
96 }
97
98 Gl::initialized = true;
99 INFO("{}: initialized", Gl::subsystem_name);
100 return {};
101}

177GLenum Gl::_check_error(const char *file, int line)
178{
179 GLenum errorCode;
180 while ((errorCode = glGetError()) != GL_NO_ERROR)
181 {
182 std::string error = "";
183 switch (errorCode)
184 {
185 case GL_INVALID_ENUM: error = "invalid enum"; break;
186 case GL_INVALID_VALUE: error = "invalid value"; break;
187 case GL_INVALID_OPERATION: error = "invalid operation"; break;
188 case GL_STACK_OVERFLOW: error = "stack overflow"; break;
189 case GL_STACK_UNDERFLOW: error = "stack underflow"; break;
190 case GL_OUT_OF_MEMORY: error = "out of memory"; break;
191 case GL_INVALID_FRAMEBUFFER_OPERATION:
192 error = "invalid framebuffer operation"; break;
193 default: error = "unknown error"; break;
194 }
195
196 error += " | " + std::string(file) + " (" + std::to_string(line) + ")";
197 ERROR("{}: {}", Gl::subsystem_name, error);
198 }
199 return errorCode;
200}

206void APIENTRY glDebugOutput([[maybe_unused]] GLenum source,
207 [[maybe_unused]] GLenum type,
208 [[maybe_unused]] GLuint id,
209 [[maybe_unused]] GLenum severity,
210 [[maybe_unused]] GLsizei length,
211 [[maybe_unused]] const GLchar *message,
212 [[maybe_unused]] const void *userParam)
213{
214 // ignore non-significant error/warning codes
215 if(id == 131169 || id == 131185 || id == 131218 || id == 131204) return;
216
217 std::stringstream out;
218 out << "Debug message (" << id << "): " << message << std::endl;
219
220 switch (source)
221 {
222 case GL_DEBUG_SOURCE_API:
223 out << "Source: API"; break;
224 case GL_DEBUG_SOURCE_WINDOW_SYSTEM:
225 out << "Source: Window System"; break;
226 case GL_DEBUG_SOURCE_SHADER_COMPILER:
227 out << "Source: Shader Compiler"; break;
228 case GL_DEBUG_SOURCE_THIRD_PARTY:
229 out << "Source: Third Party"; break;
230 case GL_DEBUG_SOURCE_APPLICATION:
231 out << "Source: Application"; break;
232 case GL_DEBUG_SOURCE_OTHER:
233 out << "Source: Other"; break;
234 } out << std::endl;
235
236 switch (type)
237 {
238 case GL_DEBUG_TYPE_ERROR:
239 out << "Type: Error"; break;
240 case GL_DEBUG_TYPE_DEPRECATED_BEHAVIOR:
241 out << "Type: Deprecated Behaviour"; break;
242 case GL_DEBUG_TYPE_UNDEFINED_BEHAVIOR:
243 out << "Type: Undefined Behaviour"; break;
244 case GL_DEBUG_TYPE_PORTABILITY:
245 out << "Type: Portability"; break;
246 case GL_DEBUG_TYPE_PERFORMANCE:
247 out << "Type: Performance"; break;
248 case GL_DEBUG_TYPE_MARKER:
249 out << "Type: Marker"; break;
250 case GL_DEBUG_TYPE_PUSH_GROUP:
251 out << "Type: Push Group"; break;
252 case GL_DEBUG_TYPE_POP_GROUP:
253 out << "Type: Pop Group"; break;
254 case GL_DEBUG_TYPE_OTHER:
255 out << "Type: Other"; break;
256 } out << std::endl;
257
258 switch (severity)
259 {
260 case GL_DEBUG_SEVERITY_HIGH:
261 out << "Severity: high"; break;
262 case GL_DEBUG_SEVERITY_MEDIUM:
263 out << "Severity: medium"; break;
264 case GL_DEBUG_SEVERITY_LOW:
265 out << "Severity: low"; break;
266 case GL_DEBUG_SEVERITY_NOTIFICATION:
267 out << "Severity: notification"; break;
268 } out << std::endl;
269 out << std::endl;
270
271 ERROR("{}: {}", Gl::subsystem_name, out.str());
272}
