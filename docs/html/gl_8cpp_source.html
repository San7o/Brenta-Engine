19std::expected<void, std::string> gl::initialize()
20{
21 GLADloadproc loadproc = (GLADloadproc) window::get_proc_address();
22 if (!gladLoadGLLoader(loadproc))
23 {
24 ERROR("Failed to initialize GLAD");
25 return std::unexpected(this->subsystem_name);
26 }
27
28 int width = window::get_width();
29 int height = window::get_height();
30
31 glViewport(0, 0, width, height);
32
33 if (enable_depth_test)
34 {
35 glEnable(GL_DEPTH_TEST);
36 INFO("GL_DEPTH_TEST enabled");
37 }
38
39 if (enable_blending)
40 {
41 glEnable(GL_BLEND);
42 glBlendFunc(GL_SRC_ALPHA, GL_ONE_MINUS_SRC_ALPHA);
43 INFO("GL_BLEND enabled (transparency)");
44 }
45
46 if (enable_cull_face)
47 {
48 glEnable(GL_CULL_FACE);
49 INFO("GL_CULL_FACE enabled (draw only visible triangles)");
50 }
51
52 if (enable_multisample)
53 {
54 glEnable(GL_MULTISAMPLE);
55 INFO("GL_MULTISAMPLE enabled");
56 }
57
58 GLenum errcode = gl::check_error();
59 if (!errcode)
60 INFO("gl initialized");
61
62 return {};
63}

121GLenum gl::_check_error(const char *file, int line)
122{
123 GLenum errorCode;
124 while ((errorCode = glGetError()) != GL_NO_ERROR)
125 {
126 std::string error;
127 switch (errorCode)
128 {
129 case GL_INVALID_ENUM:
130 error = "INVALID_ENUM";
131 break;
132 case GL_INVALID_VALUE:
133 error = "INVALID_VALUE";
134 break;
135 case GL_INVALID_OPERATION:
136 error = "INVALID_OPERATION";
137 break;
138 case GL_STACK_OVERFLOW:
139 error = "STACK_OVERFLOW";
140 break;
141 case GL_STACK_UNDERFLOW:
142 error = "STACK_UNDERFLOW";
143 break;
144 case GL_OUT_OF_MEMORY:
145 error = "OUT_OF_MEMORY";
146 break;
147 case GL_INVALID_FRAMEBUFFER_OPERATION:
148 error = "INVALID_FRAMEBUFFER_OPERATION";
149 break;
150 default:
151 error = "UNKNOWN";
152 break;
153 }
154
155 error += " | " + std::string(file) + " (" + std::to_string(line) + ")";
156 ERROR(error);
157 }
158 return errorCode;
159}
